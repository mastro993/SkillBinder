function showScanning() { els = {}; dlg.setBody(h('div', { class: 'im-center' }, spinner(16), 'Scanning…')); dlg.setFooter(cancelBtn(), h('button', { class: 'sb-btn sb-btn-primary', disabled: true }, 'Import skills')); } function showError() { els = {}; dlg.setBody(h('div', { class: 'im-center is-error' }, h('span', { class: 'im-err-ic', html: icon('alert-circle', 16) }), "Couldn't scan: ~/code/api-server can't be read.")); dlg.setFooter(cancelBtn('Close'), h('button', { class: 'sb-btn sb-btn-primary', 'data-autofocus': '', onclick: () => { S.state = 'list'; drawStates(); drawHeader(); scan('list'); } }, 'Retry')); } function showNone() { els = {}; dlg.setBody(h('div', { class: 'im-center' }, 'No new skills found.')); dlg.setFooter(cancelBtn('Close', 'sb-btn-primary')); } function showList(state) { S.skills = copy(sourceOf(state)).filter((k) => !S.imported.has(k.slug)); if (!S.skills.length) { showNone(); return; } S.choice = {}; S.skills.forEach((k) => { S.choice[k.slug] = FOUND.preferred(k); }); Object.assign(S, { ticks: FOUND.initialTicks(S.skills, S.choice), expanded: new Set(), ack: false, query: '', busy: new Set(), refreshing: false, note: false, changed: null, focus: S.skills[0].slug }); let target = null; if (state === 'versions') { ['gh-stack', 'caveman'].forEach((s) => S.expanded.add(s)); S.focus = 'gh-stack'; } if (state === 'invalid' && S.skills.some((k) => k.slug === 'legacy-deploy')) { S.ticks.add('legacy-deploy'); S.focus = target = 'legacy-deploy'; } drawList(target); } // The list's frame for the current variant; switching variants keeps ticks, versions and search. function drawList(target) { const search = searchInput({ placeholder: 'Search found skills', value: S.query, onInput: (q) => { S.query = q; update(); } }); els = { status: h('span', { class: 'im-status' }, `${S.skills.length} found · ${FOUND.inLibrary + S.imported.size} already in library`), note: h('div', { class: 'im-note', hidden: true }, ic('refresh-03', 14), 'The list changed since the scan. Check it and import again.'), list: h('div', { class: `sb-card im-list${S.variant === 'b' ? ' im-compact' : ''}` }), right: S.variant === 'b' ? h('div', { class: 'im-right' }) : null, }; const tools = h('div', { class: 'im-tools' }, search, els.status); dlg.setBody(els.right ? h('div', { class: 'im-split' }, h('div', { class: 'im-left' }, tools, els.note, els.list), els.right) : [tools, els.note, els.list]); update(); const row = target && els.list.querySelector(`[data-slug="${target}"]`); if (row) els.list.scrollTop = row.offsetTop - 80; } // ---- selection -------------------------------------------------------------------------------- function setTick(slug, on) { if (on) S.ticks.add(slug); else S.ticks.delete(slug); update(); } // Select all ticks every importable skill, up to 500, including rows the search hides. function tickAll(list, on) { if (on) { for (const k of list.filter(canTick)) { if (S.ticks.size >= FOUND.MAX) break; S.ticks.add(k.slug); } } else list.forEach((k) => S.ticks.delete(k.slug)); update(); } // Choosing another version prepares the plan again: the row is busy briefly and every tick stays. function choose(slug, id) { if (S.choice[slug] === id) return; S.choice[slug] = id; S.busy.add(slug); update(); setTimeout(() => { S.busy.delete(slug); update(); }, 600); } // ---- list ---------------------------------------------------------------------------------------- function update() { if (!dlg || !els.list) return; els.note.hidden = !S.note; els.list.classList.toggle('is-busy', S.refreshing); const top = els.list.scrollTop; const shown = S.skills.filter(matches); if (S.variant === 'b' && shown.length && !shown.some((k) => k.slug === S.focus)) S.focus = shown[0].slug; const rows = !shown.length ? 'Invalid' : v.validation === 'blocked' ? 'Blocked' : null] .filter(Boolean).join(' · '); return h('div', { class: 'im-ver' }, h('label', { class: 'im-ver-top' }, h('input', { type: 'radio', class: 'im-radio', name: `ver-${k.slug}`, checked: S.choice[k.slug] === v.id, disabled: !FOUND.importable(v) || S.busy.has(k.slug), onchange: () => choose(k.slug, v.id) }), h('span', label), v.matchesSource ? h('span', { class: 'sb-tag sb-tag-outline' }, 'Matches source') : null), details(v)); } function versions(k) { const open = S.expanded.has(k.slug); return [ h('button', { class: 'sb-btn sb-btn-xs im-vtoggle', 'aria-expanded': String(open), onclick: () => { if (!S.expanded.delete(k.slug)) S.expanded.add(k.slug); update(); } }, ic('chevron-right', 13), `${k.versions.length} versions`), open ? h('div', { class: 'im-versions' }, k.versions.map((v) => versionChoice(k, v))) : h('div', { class: 'im-lines' }, chosen(k).messages.map((m) => warn(m)), replaceLines(chosen(k))), ]; } // A and C: a flat row with every detail inline; a click on the row ticks it. function row(k) { const v = chosen(k); const ok = canTick(k); const busy = S.busy.has(k.slug); return h('div', { class: `sb-row${ok ? '' : ' is-dim'}${busy ? ' is-busy' : ''}${S.changed === k.slug ? ' is-changed' : ''}`, 'data-slug': k.slug, onclick: (e) => { if (ok && !e.target.closest('input, button, label, .im-versions')) setTick(k.slug, !S.ticks.has(k.slug)); }, }, tick(k), h('div', { class: 'sb-row-main' }, h('div', { class: 'sb-row-title' }, h('span', { class: 'sb-slug' }, k.slug), tag(v), busy ? spinner(13) : null), k.desc ? h('div', { class: 'sb-desc' }, k.desc) : null, k.versions.length > 1 ? versions(k) : details(v))); }
